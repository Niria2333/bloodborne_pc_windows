cmake_minimum_required(VERSION 3.24)
project(bbport_windows LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_C_COMPILER_ID STREQUAL "GNU")
    message(FATAL_ERROR "Use Windows x64 MinGW GCC with winpthreads. Linux builds use build.sh.")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_compile_definitions(_WIN32_WINNT=0x0A00 WINVER=0x0A00 NOMINMAX _FILE_OFFSET_BITS=64)
add_compile_options(-fno-omit-frame-pointer)
find_package(Vulkan REQUIRED)
find_package(PkgConfig REQUIRED)
find_package(SDL3 CONFIG REQUIRED)
file(GLOB ATRAC9 CONFIGURE_DEPENDS third_party/LibAtrac9/C/src/*.c)
add_library(atrac9 STATIC ${ATRAC9})
target_compile_options(atrac9 PRIVATE -w)
file(GLOB RUNTIME CONFIGURE_DEPENDS src/runtime*.c)
add_library(bbruntime STATIC ${RUNTIME})
target_include_directories(bbruntime PUBLIC . src)
target_link_libraries(bbruntime PUBLIC SDL3::SDL3 winpthread bcrypt ws2_32)
target_compile_options(bbruntime PRIVATE -Wall -Wextra)
option(BB_DIAGNOSTIC_ONLY "CPU/Vulkan diagnostics only; cannot play the game" OFF)
if(BB_DIAGNOSTIC_ONLY)
    add_library(bbgpu_diagnostic STATIC tests/windows_gpu_diagnostic.c)
    target_include_directories(bbgpu_diagnostic PUBLIC .)
    set(BB_GPU_TARGET bbgpu_diagnostic)
else()
    add_subdirectory(gpu)
    set(BB_GPU_TARGET bbgpu)
endif()
add_executable(bb-probe src/probe.c src/vulkan_smoke.c)
target_link_libraries(bb-probe PRIVATE bbruntime ${BB_GPU_TARGET} atrac9 Vulkan::Vulkan)
include(CTest)
if(BUILD_TESTING)

    target_compile_options(trainer-fixture-windows PRIVATE -Wall -Wextra -Werror)
